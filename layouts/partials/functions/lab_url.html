{{- /* Resolve a URL written in page front matter. External URLs (http…, mailto:) pass through. Site paths are written without a leading slash ("research/", "publication/2026-oakland/", "research/#ai-security"); they must match a page in content/ (otherwise the build fails). The result is that page's own permalink (plus any #fragment), so folder casing or content paths such as "authors/Prof-YuanTian/" still produce the published URL, which matters on case-sensitive hosts like GitHub Pages and under the baseURL subpath. */ -}} {{- $url := . | default "" -}} {{- if not $url -}} {{- errorf "lab_url: a link in the page front matter has no url" -}} {{- end -}} {{- $out := $url -}} {{- if not (or (strings.HasPrefix $url "http") (strings.HasPrefix $url "mailto:")) -}} {{- $parsed := urls.Parse $url -}} {{- with site.GetPage (printf "/%s" (strings.Trim $parsed.Path "/")) -}} {{- $out = .RelPermalink -}} {{- with $parsed.Fragment -}} {{- $out = printf "%s#%s" $out . -}} {{- end -}} {{- else -}} {{- errorf "lab_url: %q does not match any page in content/; fix this url in the page front matter" $url -}} {{- end -}} {{- end -}} {{- return $out -}}
